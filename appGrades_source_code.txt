    # ========================================================
    # GRADE PARSER
    # ========================================================

    def parse_grades(self, path):

        """
        Expected grades.txt format:

        Andrae John V. Almodiente, BSCS 2
        CSCC 102, Fundumentals of Programming, 3, 1.25
        CSMath 101, Trigonometric Mathematics, 3, 2.00
        AH 111, Purposive Communication, 3, 2.25
        AH 112, Arts Apprectiation, 3, 2.50
        CSPC 102, Discrete Structures II, 3, 1.25
        WKL 2, PananalikSox, 3, 1.75
        PathFit 2, Physical Education, 3, 1.00
        RE 112, life of Jesus Christ, 3, 1.50
        NSTP 2, National Service Program, 3, 2.50
        SDS 2, Students Development, 1, 1.00
        """

        records = []

        with open(
            path,
            "r",
            encoding="utf-8"
        ) as f:

            lines = [
                line.strip()
                for line in f
                if line.strip()
                and not line.strip().startswith("#")
            ]

        if not lines:
            return records

        # ----------------------------------------------------
        # FIRST LINE
        # Student Name, Program & Year
        # ----------------------------------------------------

        student_info = [
            p.strip()
            for p in lines[0].split(",")
        ]

        if len(student_info) >= 2:

            self.student_name = student_info[0]

            self.program_year = student_info[1]

            self.student_var.set(
                self.student_name
            )

            self.program_var.set(
                self.program_year
            )

        # ----------------------------------------------------
        # COURSE RECORDS
        # ----------------------------------------------------

        for line in lines[1:]:

            parts = [
                p.strip()
                for p in line.split(",")
            ]

            if len(parts) < 4:
                continue

            code = parts[0]
            title = parts[1]
            units = parts[2]
            grade = parts[3]

            if (
                self.is_number(units)
                and self.is_number(grade)
            ):

                records.append(
                    {
                        "code": code,
                        "title": title,
                        "units": float(units),
                        "grade": float(grade)
                    }
                )

        return records


    # ========================================================
    # NUMBER CHECK
    # ========================================================

    @staticmethod
    def is_number(value):

        try:
            float(value)
            return True

        except ValueError:
            return False


    # ========================================================
    # LOAD GRADES
    # ========================================================

    def load_grades(self):

        path = self.file_var.get()

        # ----------------------------------------------------
        # Find grades.txt
        # ----------------------------------------------------

        if not os.path.exists(path):

            selected = filedialog.askopenfilename(
                title="Select grades.txt",
                filetypes=[
                    ("Text files", "*.txt"),
                    ("All files", "*.*")
                ]
            )

            if not selected:

                self.load_sample_data()

                self.log_message(
                    "READY   -> Sample data displayed."
                )

                return

            path = selected

            self.file_var.set(
                path
            )

        # ----------------------------------------------------
        # Read file
        # ----------------------------------------------------

        try:

            records = self.parse_grades(
                path
            )

            if not records:

                raise ValueError(
                    "No valid grade records were found."
                )

            self.display_records(
                records
            )

            self.log_message(
                f"LOADED  -> {len(records)} course(s) for "
                f"{self.student_name}"
            )

        except Exception as exc:

            messagebox.showerror(
                "Grade Loading Error",
                str(exc)
            )

            self.load_sample_data()

            self.log_message(
                "ERROR   -> Could not read grades.txt."
            )

    # ========================================================
    # DISPLAY GRADES
    # ========================================================

    def display_records(self, records):

        # Clear existing rows

        for item in self.tree.get_children():

            self.tree.delete(
                item
            )

        total_units = 0

        weighted_points = 0

        # ----------------------------------------------------
        # Add records
        # ----------------------------------------------------

        for record in records:

            units = float(
                record["units"]
            )

            grade = float(
                record["grade"]
            )

            self.tree.insert(
                "",
                "end",
                values=(
                    record["code"],
                    record["title"],
                    f"{units:g}",
                    f"{grade:.2f}"
                )
            )

            total_units += units

            weighted_points += (
                units * grade
            )

        # ----------------------------------------------------
        # Calculate average
        # ----------------------------------------------------

        average = (
            weighted_points / total_units
            if total_units
            else 0
        )

        self.total_units_var.set(
            f"{total_units:g}"
        )

        self.average_var.set(
            f"{average:.2f}"
        )

    def send_report(self):

        recipient = (
            self.recipient_var.get()
            .strip()
        )

        # ----------------------------------------------------
        # Validate recipient
        # ----------------------------------------------------

        if not recipient:

            messagebox.showwarning(
                "Missing Recipient",
                "Enter an email address."
            )

            return

        if not re.match(
            r"^[^@\s]+@[^@\s]+\.[^@\s]+$",
            recipient
        ):

            messagebox.showwarning(
                "Invalid Email",
                "Enter a valid email address."
            )

            return

        # ----------------------------------------------------
        # Check API key
        # ----------------------------------------------------

        if not BREVO_API_KEY:

            messagebox.showinfo(
                "Brevo API Key Required",
                "The GUI is ready, but BREVO_API_KEY "
                "is not configured.\n\n"
                "Windows PowerShell:\n\n"
                '$env:BREVO_API_KEY="your_api_key_here"\n\n'
                "Also set BREVO_SENDER_EMAIL to your "
                "verified Brevo sender email."
            )

            self.log_message(
                "WAITING -> Brevo API key is not configured."
            )

            return

        # ----------------------------------------------------
        # Sending
        # ----------------------------------------------------

        self.log_message(
            f"SENDING -> Grade report to {recipient}..."
        )

        payload = {

            "sender": {
                "name": "NDMU Grade Report Console",
                "email": DEFAULT_SENDER_EMAIL
            },

            "to": [
                {
                    "email": recipient,
                    "name": self.student_name
                }
            ],

            "subject": (
                f"Grade Report - "
                f"{self.student_name}"
            ),

            "htmlContent": (
                self.create_email_body()
            )
        }

        request = urllib.request.Request(

            "https://api.brevo.com/v3/smtp/email",

            data=json.dumps(
                payload
            ).encode("utf-8"),

            headers={
                "accept": "application/json",
                "api-key": BREVO_API_KEY,
                "content-type": "application/json"
            },

            method="POST"
        )

        # ----------------------------------------------------
        # Send request
        # ----------------------------------------------------

        try:

            with urllib.request.urlopen(
                request,
                timeout=20
            ) as response:

                response.read()

            self.log_message(
                f"SENT    -> Successfully delivered "
                f"to {recipient}"
            )

            messagebox.showinfo(
                "Report Sent",
                "The grade report was successfully "
                f"sent to:\n{recipient}"
            )

        # ----------------------------------------------------
        # Brevo error
        # ----------------------------------------------------

        except urllib.error.HTTPError as exc:

            details = (
                exc.read()
                .decode(
                    "utf-8",
                    errors="replace"
                )
            )

            self.log_message(
                f"FAILED  -> Brevo HTTP "
                f"{exc.code}: {details}"
            )

            messagebox.showerror(
                "Brevo Error",
                f"Brevo returned HTTP {exc.code}.\n\n"
                f"{details}"
            )

        # ----------------------------------------------------
        # Other error
        # ----------------------------------------------------

        except Exception as exc:

            self.log_message(
                f"FAILED  -> {exc}"
            )

            messagebox.showerror(
                "Sending Error",
                str(exc)
            )
